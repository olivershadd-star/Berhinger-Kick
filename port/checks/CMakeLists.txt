cmake_minimum_required(VERSION 3.22)
project(KickPlatformChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
get_filename_component(REPO_ROOT "${PORT_ROOT}/.." ABSOLUTE)
set(AUDIO_ROOT "${REPO_ROOT}/upstream/0700-AAM/AudIO")
enable_testing()
add_executable(fixed_block_bridge "${PORT_ROOT}/tests/fixed_block_bridge.cpp")
target_include_directories(fixed_block_bridge PRIVATE "${PORT_ROOT}/modules/AudIO_juce" "${AUDIO_ROOT}")
add_test(NAME device_buffer_continuity COMMAND fixed_block_bridge)
option(KICK_CHECK_AUDIO_DEVICE "Compile the Windows JUCE device bridge (not the KICK application)" OFF)
set(KICK_JUCE_SOURCE "" CACHE PATH "Local JUCE 7.0.12 checkout")
if(KICK_CHECK_AUDIO_DEVICE)
    if(NOT WIN32)
        message(FATAL_ERROR "The device adapter compile check is for Windows")
    endif()
    if(NOT KICK_JUCE_SOURCE)
        include(FetchContent)
        FetchContent_Declare(juce_headers
            GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
            GIT_TAG 7.0.12 GIT_SHALLOW TRUE)
        FetchContent_GetProperties(juce_headers)
        if(NOT juce_headers_POPULATED)
            FetchContent_Populate(juce_headers)
        endif()
        set(KICK_JUCE_SOURCE "${juce_headers_SOURCE_DIR}")
    endif()
    add_library(audio_device_compile OBJECT "${PORT_ROOT}/tests/audio_device_compile.cpp")
    target_include_directories(audio_device_compile PRIVATE "${PORT_ROOT}/modules" "${AUDIO_ROOT}" "${KICK_JUCE_SOURCE}/modules")
    target_compile_definitions(audio_device_compile PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1
        JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
endif()
message(STATUS "Platform checks only: this project does not produce Kick.exe")
